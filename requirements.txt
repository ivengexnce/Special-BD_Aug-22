#  A cinematic birthday universe built for Aashu 🌸
# ══════════════════════════════════════════════════════════════════════════════
#
#  This is a Node.js / Next.js project.
#  There is no pip install — use:
#
#    npm install       ← installs all dependencies from package.json
#    npm run dev       ← starts local dev server at http://localhost:3000
#    npm run build     ← creates production build
#    npm start         ← runs production server
#
# ── Runtime: Node.js ≥ 18.x recommended ──────────────────────────────────────

# ── Framework ─────────────────────────────────────────────────────────────────
next@16.3.1                      # App router, server components, image optimisation
react@19.2.8                     # UI library
react-dom@19.2.8                 # DOM bindings for React

# ── 3D / WebGL ────────────────────────────────────────────────────────────────
three@^0.185.1                   # 3D engine (WebGL)
@react-three/fiber@^9.7.0        # React renderer for Three.js
@react-three/drei@^10.7.8        # Helpers & abstractions for R3F
@react-three/postprocessing@^3.0.5  # Bloom, chromatic aberration, etc.
postprocessing@^6.39.4           # Post-processing effects (peer dep)

# ── Animation ─────────────────────────────────────────────────────────────────
gsap@^3.15.0                     # GSAP — scroll triggers, timelines, typewriter
@gsap/react@^2.1.2               # React hooks for GSAP
framer-motion@^13.1.0            # Motion components (AnimatePresence, motion.div)
